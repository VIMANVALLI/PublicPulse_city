import { useEffect, useState } from "react";
import api from "../../services/api";
import { Bar } from "react-chartjs-2";
import {Chart as ChartJS,BarElement,CategoryScale,LinearScale,Tooltip,Legend,} from "chart.js";
ChartJS.register(BarElement, CategoryScale, LinearScale, Tooltip, Legend);
export default function ViewGraph() {
  const [counts, setCounts] = useState({ positive: 0, negative: 0, neutral: 0 });
  useEffect(() => {
    const fetchData = async () => {
      try {
        const { data } = await api.get("/sentiment-counts");
        setCounts(data);
      } catch {}
    };
    fetchData();
    window.addEventListener("opinionAdded", fetchData);
    return () => window.removeEventListener("opinionAdded", fetchData);
  }, []);
  const chartData = {
    labels: ["Positive", "Negative", "Neutral"],
    datasets: [
      {
        label: "Sentiment Count",
        data: Object.values(counts),
        backgroundColor: ["#22c55e", "#ef4444", "#f59e0b"],
      },
    ],
  };
  return (
    <div className="ml-64 p-6">
      <h2 className="text-2xl font-bold mb-6">Sentiment Graph</h2>
      <div className="bg-white p-6 rounded shadow w-[600px]">
        <Bar data={chartData} />
      </div>
    </div>
  );
}
